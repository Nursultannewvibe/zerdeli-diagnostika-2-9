/**
 * Zerdeli — приёмник заявок и результатов диагностики.
 *
 * Один адрес на всё: страница шлёт сюда заявку вместе с результатом теста.
 * Скрипт кладёт строки в таблицу, собирает PDF-отчёт и пишет в телеграм-бот.
 * Токен бота остаётся здесь, на стороне Google, — в коде сайта его нет,
 * иначе его вытащил бы любой, кто открыл исходник страницы.
 *
 * Как поставить:
 *   1. Открыть нужную Google-таблицу → Расширения → Apps Script
 *   2. Стереть всё, что там есть, вставить этот файл
 *   3. Заполнить настройки ниже
 *   4. Запустить функцию proverka (кнопка ▶) — Google спросит разрешения,
 *      после этого придёт сообщение в бот и появятся строки в таблице
 *   5. Развернуть → Новое развёртывание → Веб-приложение
 *        Запуск от имени: я
 *        Доступ: все
 *      скопировать адрес (…/exec) и вставить его в landing.js
 *
 * После правки кода развёртывание нужно обновить:
 *   Развернуть → Управление развёртываниями → карандаш → Версия: новая → Развернуть.
 *   Адрес при этом не меняется.
 */

/* ================== НАСТРОЙКИ ================== */

// Для группы — id группы, он с минусом: -1001234567890.
// Можно несколько через запятую: '123456789,-1001234567890'

// Оставить пустым, если скрипт открыт из самой таблицы (Расширения → Apps Script).
// Иначе — id таблицы из её адреса: docs.google.com/spreadsheets/d/ЭТОТ_КУСОК/edit

/* ================== ПРИЁМ ================== */

function doPost(e) {
  var lock = LockService.getScriptLock();
  try { lock.waitLock(20000); } catch (err) { /* пишем всё равно */ }
  try {
    var d = JSON.parse(e.postData.contents);

    // Заявка и результат приходят одной посылкой: контакт спрашивают уже
    // после теста. Но приходят и по отдельности — второй предмет идёт без
    // заявки, а старая форма шлёт заявку без теста. Смотрим по полям.
    var estZayavka = !!(d.when || d.source);
    var estResultat = !!d.testId;

    if (estResultat) { d.pdfUrl = sobratPdf(d); }
    if (estZayavka) { zayavkaRow(d); }
    if (estResultat) { rezultatRow(d); }

    telegram(soobshenie(d, estZayavka, estResultat));
    return otvet({ ok: true, pdf: d.pdfUrl || '' });
  } catch (err) {
    return otvet({ ok: false, error: String(err) });
  } finally {
    try { lock.releaseLock(); } catch (err) {}
  }
}

/**
 * Ссылка «Скачать отчёт» с сайта ведёт сюда: …/exec?pdf=КОД.
 * Файл обычно уже готов — его собрали в момент приёма результата.
 * Если нет, собираем сейчас: лучше подождать пару секунд, чем пустая страница.
 */
function doGet(e) {
  var kod = e && e.parameter && e.parameter.pdf;
  if (!kod) return otvet({ ok: true, service: 'zerdeli-diagnostika' });

  var row = naytiPoKodu(kod);
  if (!row) return stranica('Отчёт не найден',
    'Похоже, ссылка устарела или в ней опечатка. Напишите нам в WhatsApp — пришлём отчёт вручную.',
    'https://wa.me/' + WHATSAPP);

  var url = row.pdfUrl || sobratPdf(row);
  if (!url) return stranica('Отчёт готовится',
    'Обновите страницу через несколько секунд.', '');
  if (!row.pdfUrl) zapisatPdfUrl(kod, url);

  return stranica('Открываем отчёт…', 'Если ничего не произошло, нажмите на ссылку.', url);
}

function otvet(o) {
  return ContentService.createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}

// маленькая страница-прослойка: сама уводит на файл, ссылка — на случай,
// если браузер запретил переход
function stranica(zagolovok, text, url) {
  var h = '<!doctype html><meta charset="utf-8">' +
    '<div style="font:16px/1.5 system-ui,sans-serif;max-width:32em;margin:14vh auto;padding:0 20px;text-align:center">' +
    '<h2 style="font-size:22px">' + esc(zagolovok) + '</h2><p style="color:#555">' + esc(text) + '</p>' +
    (url ? '<p><a href="' + esc(url) + '" style="color:#6F8F05;font-weight:600">Открыть отчёт</a></p>' +
           '<script>setTimeout(function(){(window.top||window).location.href=' +
           JSON.stringify(url) + ';},400)<\/script>' : '') +
    '</div>';
  return HtmlService.createHtmlOutput(h)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/* ================== СТРОКИ В ТАБЛИЦЕ ================== */

function zayavkaRow(d) {
  var list = list_(LIST_ZAYAVKI, [
    'Дата', 'Имя', 'Телефон', 'Класс', 'Когда удобно',
    'Язык страницы', 'Страница', 'Источник'
  ]);
  list.appendRow([
    data_(d.date), d.name || '', d.phone || '', d.grade || '',
    kogda_(d.when), d.pageLang || '', d.page || '', d.source || ''
  ]);
}

var STOLBCY_REZULTATA = [
  'Дата', 'Имя', 'Телефон', 'Класс', 'Предмет', 'Тест',
  'Верно', 'Из', 'Процент', 'По темам', 'Ответы', 'Язык', 'Страница',
  'Код отчёта', 'PDF'
];

function rezultatRow(d) {
  var list = list_(LIST_REZULTATY, STOLBCY_REZULTATA);
  list.appendRow([
    data_(d.date), d.name || '', d.phone || '', d.grade || '', d.subject || '',
    d.testId || '', d.right, d.scored, d.percent,
    temy_(d.topics), otvety_(d.answers), d.lang || '', d.page || '',
    d.uid || '', d.pdfUrl || ''
  ]);
}

// строка результата по коду отчёта — для ссылки «Скачать отчёт»
function naytiPoKodu(kod) {
  var list = kniga_().getSheetByName(LIST_REZULTATY);
  if (!list || list.getLastRow() < 2) return null;
  var iKod = STOLBCY_REZULTATA.indexOf('Код отчёта');
  var vse = list.getRange(2, 1, list.getLastRow() - 1, STOLBCY_REZULTATA.length).getValues();
  for (var i = vse.length - 1; i >= 0; i--) {      // с конца: свежие вероятнее
    if (String(vse[i][iKod]) !== kod) continue;
    var o = {};
    for (var k = 0; k < STOLBCY_REZULTATA.length; k++) o[STOLBCY_REZULTATA[k]] = vse[i][k];
    return {
      stroka: i + 2,
      date: o['Дата'], name: o['Имя'], phone: o['Телефон'], grade: o['Класс'],
      subject: o['Предмет'], testId: o['Тест'], right: o['Верно'], scored: o['Из'],
      percent: o['Процент'], topics: o['По темам'], lang: o['Язык'],
      uid: o['Код отчёта'], pdfUrl: o['PDF']
    };
  }
  return null;
}

function zapisatPdfUrl(kod, url) {
  var row = naytiPoKodu(kod);
  if (!row) return;
  kniga_().getSheetByName(LIST_REZULTATY)
    .getRange(row.stroka, STOLBCY_REZULTATA.indexOf('PDF') + 1).setValue(url);
}

/* ================== PDF-ОТЧЁТ ================== */

/**
 * Отчёт собирается из обычного HTML: Google сам переводит его в PDF,
 * и кириллица с казахскими буквами выходят как надо. Файл кладём на Диск
 * и открываем по ссылке — так он скачивается с любого телефона.
 */
function sobratPdf(d) {
  try {
    var html = otchetHtml(d);
    var blob = Utilities.newBlob(html, 'text/html', 'otchet.html')
      .getAs('application/pdf')
      .setName(imyaFayla(d));
    var file = papka_().createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch (err) {
    return '';   // отчёт — приятное дополнение, из-за него терять заявку нельзя
  }
}

function imyaFayla(d) {
  var kogda = Utilities.formatDate(data_(d.date), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  return 'Диагностика — ' + (d.name || 'ученик') + ' — ' +
    (d.subject || '') + ' — ' + kogda + '.pdf';
}

function papka_() {
  var it = DriveApp.getFoldersByName(PAPKA_OTCHETOV);
  return it.hasNext() ? it.next() : DriveApp.createFolder(PAPKA_OTCHETOV);
}

function cvet_(pct) {

}

/* ================== ТЕЛЕГРАМ ================== */

function soobshenie(d, estZayavka, estResultat) {
  var t = '';
  if (estZayavka && estResultat) t += '<b>Заявка + тест пройден</b>\n';
  else if (estZayavka) t += '<b>Заявка на диагностику</b>\n';
  else t += '<b>Тест пройден</b>\n';

  t += '<b>' + esc(d.name) + '</b> · ' + esc(d.phone) + '\n';

  if (estResultat) {
    t += esc(d.grade) + ' класс · ' + esc(d.subject) + '\n';
    t += 'Результат: <b>' + d.percent + '%</b> (' + d.right + ' из ' + d.scored + ')';
    t += probely_(d.topics);
    if (d.pdfUrl) t += '\n\n<a href="' + esc(d.pdfUrl) + '">Отчёт PDF</a>';
  }
  if (estZayavka) {
    t += (estResultat ? '\n\n' : '') + 'Когда удобно: ' + esc(kogda_(d.when)) +
         '\nЯзык: ' + (d.pageLang === 'kz' ? 'казахский' : 'русский');
  }
  return t;
}

function telegram(text) {
  if (!TG_TOKEN || !TG_CHAT) return;
  var chats = String(TG_CHAT).split(',');
  for (var i = 0; i < chats.length; i++) {
    var chat = chats[i].trim();
    if (!chat) continue;
    try {
      UrlFetchApp.fetch('https://api.telegram.org/bot' + TG_TOKEN + '/sendMessage', {
        method: 'post',
        payload: { chat_id: chat, text: text, parse_mode: 'HTML',
                   disable_web_page_preview: 'true' },
        muteHttpExceptions: true
      });
    } catch (err) {
      // молчим: сообщение в бот не должно ломать запись в таблицу
    }
  }
}

/* ================== МЕЛОЧИ ================== */

function kniga_() {
  return SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActive();
}

// лист по имени: нет — заводим и ставим шапку
function list_(name, shapka) {
  var kn = kniga_();
  var l = kn.getSheetByName(name);
  if (!l) {
    l = kn.insertSheet(name);
    l.appendRow(shapka);
    l.getRange(1, 1, 1, shapka.length).setFontWeight('bold');
    l.setFrozenRows(1);
  }
  return l;
}

// строка вида 2026-09-05T12:34:56.000Z → нормальная дата в таблице
function data_(iso) {
  var d = iso ? new Date(iso) : new Date();
  return isNaN(d.getTime()) ? new Date() : d;
}

// значения из выпадающего списка на лендинге — в человеческий вид
function kogda_(v) {
  var slovar = {
    asap: 'Как можно скорее',
    evening: 'В будни после 18:00',
    weekend: 'В выходные'
  };
  return slovar[v] || v || '';
}

// движок шлёт {тема: процент}; строку из таблицы тоже понимаем
function temySpisok_(t) {
  var out = [];
  if (!t) return out;
  if (typeof t === 'string') {
    var chasti = t.split('·');
    for (var i = 0; i < chasti.length; i++) {
      var m = chasti[i].match(/^(.*?)\s(\d+)%\s*$/);
      if (m) out.push({ name: m[1].trim(), pct: Number(m[2]) });
    }
    return out;
  }
  for (var k in t) if (t.hasOwnProperty(k)) out.push({ name: k, pct: Number(t[k]) });
  return out;
}

// в таблицу — все темы одной ячейкой: «Дроби 40% · Уравнения 85%»
function temy_(t) {
  return temySpisok_(t).map(function (p) { return p.name + ' ' + p.pct + '%'; }).join(' · ');
}

// В таблицу идут все темы, а в бот — только те, где просело: список из
// пятнадцати строк в телефоне не читают, а разговор с родителем строится
// вокруг пробелов. Порог 60% — тот же, по которому лендинг красит «пробел».
function probely_(t) {
  var vse = temySpisok_(t);
  if (!vse.length) return '';
  var plohie = vse.filter(function (p) { return p.pct < 60; });
  if (!plohie.length) return '\n\nПробелов нет — все темы выше 60%.';
  plohie.sort(function (a, b) { return a.pct - b.pct; });
  var vidno = plohie.slice(0, 8);
  var text = '\n\n<b>Пробелы:</b>\n' + vidno.map(function (p) {
    return '• ' + esc(p.name) + ' — ' + p.pct + '%';
  }).join('\n');
  if (plohie.length > vidno.length) {
    text += '\nи ещё ' + (plohie.length - vidno.length) + ' — смотри в таблице';
  }
  return text;
}

function otvety_(a) {
  if (!a) return '';
  if (typeof a === 'string') return a;
  return JSON.stringify(a);
}

function esc(s) {
  return String(s == null ? '' : s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}
